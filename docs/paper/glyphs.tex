% Characters of the Lean sources that the monospace font (DejaVu Sans Mono) is not used for: the
% subscript letters of `::ₘ`, `Sⁱₙ`, `xₖ`, drawn as lowered small letters, and the script `ℓ`,
% taken from the math font. Each is set in a box of the width of one monospace character, so
% that the columns of the listings stay aligned.
\newcommand{\fbch}[1]{\makebox[0.602em][c]{\fallbackfont #1}}
\newcommand{\subdrawn}[1]{\makebox[0.602em][c]{\raisebox{-0.38ex}{\scalebox{0.7}{#1}}}}
\newif\ifsubfont
\subfontfalse
\newcommand{\subch}[2]{\ifsubfont\makebox[0.602em][c]{\subfont #1}\else\subdrawn{#2}\fi}
\newunicodechar{ₘ}{\subch{ₘ}{m}}
\newunicodechar{ₖ}{\subch{ₖ}{k}}
\newunicodechar{ₙ}{\subch{ₙ}{n}}
\newunicodechar{ℓ}{\fbch{ℓ}}
